#!/usr/bin/env python3
"""Read-only battery/clock sampling; power_now is diagnostic, not authoritative."""
import argparse
import glob
import json
import statistics
import time
from pathlib import Path


def read(path):
    try:
        value = Path(path).read_text().strip()
        try:
            return int(value)
        except ValueError:
            return value
    except OSError:
        return None


def snapshot():
    battery = Path('/sys/class/power_supply/battery')
    result = {'boottime': time.clock_gettime(time.CLOCK_BOOTTIME),
              'timestamp': time.time()}
    for key in ('status', 'capacity', 'charge_counter', 'current_now',
                'voltage_now', 'power_now', 'temp'):
        result[key] = read(battery / key)
    current, voltage = result['current_now'], result['voltage_now']
    result['iv_watts'] = (abs(current * voltage) / 1e12
                          if isinstance(current, int) and isinstance(voltage, int) else None)
    result['cpu'] = {Path(p).name: {k: read(Path(p) / k) for k in
                     ('scaling_governor', 'scaling_cur_freq', 'scaling_max_freq')}
                     for p in glob.glob('/sys/devices/system/cpu/cpufreq/policy*')}
    result['gpu'] = {k: read('/sys/class/devfreq/3d00000.gpu/' + k)
                     for k in ('governor', 'cur_freq', 'min_freq', 'max_freq')}
    result['brightness'] = {Path(p).name: read(Path(p) / 'brightness')
                            for p in glob.glob('/sys/class/backlight/*')}
    return result


def summarize(samples):
    first, last = samples[0], samples[-1]
    seconds = last['boottime'] - first['boottime']
    watts = [s['iv_watts'] for s in samples if s['iv_watts'] is not None]
    result = {'seconds': seconds, 'samples': len(samples),
              'median_iv_watts': statistics.median(watts) if watts else None,
              'capacity_start': first['capacity'], 'capacity_end': last['capacity'],
              'charge_delta_uah': None, 'charge_based_watts': None}
    # Reject charge estimates across charging/discharging changes or a reset.
    counters = [s['charge_counter'] for s in samples]
    valid = (seconds > 0 and all(s['status'] == 'Discharging' for s in samples)
             and all(isinstance(c, int) for c in counters)
             and all(a >= b for a, b in zip(counters, counters[1:])))
    voltages = [s['voltage_now'] for s in samples if isinstance(s['voltage_now'], int)]
    if valid and voltages:
        delta = counters[0] - counters[-1]
        result['charge_delta_uah'] = delta
        result['charge_based_watts'] = delta * 3600 / seconds * statistics.mean(voltages) / 1e12
    return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--seconds', type=int, default=120)
    parser.add_argument('--interval', type=float, default=2)
    parser.add_argument('--snapshot', action='store_true')
    args = parser.parse_args()
    if args.seconds < 1 or args.interval <= 0:
        parser.error('duration and interval must be positive')
    samples = [snapshot()]
    print(json.dumps({'sample': samples[0]}), flush=True)
    if args.snapshot:
        return
    deadline = samples[0]['boottime'] + args.seconds
    while time.clock_gettime(time.CLOCK_BOOTTIME) < deadline:
        time.sleep(min(args.interval, max(0, deadline - time.clock_gettime(time.CLOCK_BOOTTIME))))
        samples.append(snapshot())
        print(json.dumps({'sample': samples[-1]}), flush=True)
    print(json.dumps({'summary': summarize(samples)}), flush=True)


if __name__ == '__main__':
    main()
