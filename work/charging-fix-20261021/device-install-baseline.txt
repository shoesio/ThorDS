# Preserve all existing storage/firmware hooks and start the splash after udev.
thords_hooks=()
for thords_hook in "${HOOKS[@]}"; do
    [[ "$thords_hook" == thords-splash ]] && continue
    thords_hooks+=("$thords_hook")
    [[ "$thords_hook" == udev ]] && thords_hooks+=(thords-splash)
done
HOOKS=("${thords_hooks[@]}")
unset thords_hooks thords_hook
# Generated on the actual Thor: trim the normal initramfs to detected hardware.
# The fallback preset explicitly skips autodetect and keeps the complete image.
# Do not include this local policy in images generated on a build VM.
if [[ " ${HOOKS[*]} " != *" autodetect "* ]]; then
    thords_boot_hooks=()
    for thords_boot_hook in "${HOOKS[@]}"; do
        thords_boot_hooks+=("$thords_boot_hook")
        [[ "$thords_boot_hook" != udev ]] || thords_boot_hooks+=(autodetect)
    done
    HOOKS=("${thords_boot_hooks[@]}")
    unset thords_boot_hooks thords_boot_hook
fi
